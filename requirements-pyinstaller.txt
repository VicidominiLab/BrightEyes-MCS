# Lean, reproducible Windows CPython 3.12-3.14 environment for freezing the
# full BrightEyes-MCS application with PyInstaller.
#
# Actual dependency minimum: CPython 3.12 (numpy==2.5.0 is the limiting pin).
# Release-tested/recommended version: CPython 3.13. The repository's installer
# and CI currently exercise only 3.13, although a complete CPython 3.12 Windows
# wheel resolution was verified on 2026-08-11.
#
# Create a dedicated environment and install only this file. Do not run a plain
# `pip install .` afterward: the dependencies in pyproject.toml would restore
# the larger general-purpose environment. If an installed distribution is
# needed, use `pip install --no-deps .` after installing this file.
#
# These versions match the environment audited on 2026-08-11, except
# nifpga-fast-fifo-recv, which is intentionally updated to the project's
# required 0.102.0 release.

# Freezer (pin it so release builds do not change unexpectedly).
PyInstaller==6.21.0

# Native acquisition backends.
brighteyes-mcs-cylibs==1.0.2
nifpga==22.0.0
nifpga-fast-fifo-recv==0.102.0

# GUI, plotting, and embedded console.
PySide6==6.11.1
pyqtgraph==0.13.7
qtconsole==5.7.2
matplotlib==3.11.0

# Core numerical, storage, and image functionality.
numpy==2.5.0
h5py==3.16.0
imageio==2.37.3
psutil==7.1.3

# REST API and client functionality imported by the main window.
fastapi==0.138.0
uvicorn==0.49.0
python-multipart==0.0.32
requests==2.34.2

# Optional built-in plugins are deliberately excluded from the base EXE:
# - ao_system needs `pyserial`.
# - the optional FLIM functions in the DFD plugin need `brighteyes-flim`.
# Add and audit those distributions only when those features will be shipped.
#
# Deliberately removed from the normal requirements:
# - brighteyes-mcs-reader: used by scripts/shift_vectors.py, not the application.
# - scipy, pyqtdarktheme, atomics, markdown, importlib-metadata: no application
#   imports found during the audit.
# - setuptools: build tooling, not a frozen runtime dependency.
# - PySide6_Addons/PySide6_Essentials: installed transitively by PySide6.
# - pywin32: not imported directly; required Windows pieces are pulled in by
#   the embedded Jupyter/Qt console dependency chain where applicable.
